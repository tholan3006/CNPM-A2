\section{Tổng quan bối cảnh, Phân công nhiệm vụ và Đặc tả Luồng Giỏ hàng (FR-04)}

\subsection{Bối cảnh phân hệ Giỏ hàng \& Đặt hàng trong BrewLite}
Trong ứng dụng gọi đồ uống trực tuyến \textbf{BrewLite}, phân hệ \textbf{"Giỏ hàng \& Khởi tạo Đơn hàng"} đóng vai trò then chốt kết nối giữa việc duyệt sản phẩm (Menu) và thực hiện thanh toán (Checkout). Phân hệ đảm nhận các giải pháp kỹ thuật cốt lõi:
\begin{itemize}
    \item \textbf{Quản lý trạng thái tức thì (Client-side State):} Sử dụng thư viện Zustand kết hợp đồng bộ hóa với \texttt{localStorage}, đảm bảo giỏ hàng không bị mất khi khách hàng vô tình tải lại trang hoặc đổi thiết bị trong cùng phiên.
    \item \textbf{Badge số lượng thời gian thực:} Tự động tính toán tổng số món hiển thị trên thanh điều hướng, giúp khách hàng nắm bắt tình trạng giỏ hàng ở bất kỳ màn hình nào.
    \item \textbf{Tính toán bảo mật phía Máy chủ (Server-side Recalculation):} Khách hàng chỉ gửi danh sách món, kích thước (Size S/M/L) và danh sách Topping. Toàn bộ giá bán gốc, phụ phí kích thước và chiết khấu Coupon được truy vấn và tính toán trực tiếp tại Backend NestJS, loại bỏ hoàn toàn nguy cơ giả mạo giá tiền từ phía máy trạm.
    \item \textbf{Kiểm soát tồn kho đồng thời (Concurrency Control):} Áp dụng cơ chế Khóa lạc quan (Optimistic Locking) thông qua cột \texttt{version} trong bảng \texttt{Product}, tự động rollback và retry tối đa 5 lần khi có nhiều khách hàng cùng tranh chấp mua những phần đồ uống cuối cùng.
\end{itemize}

\subsection{Đặc tả chi tiết Luồng nghiệp vụ tạo đơn hàng (FR-04)}
\begin{table}[h!]
\centering
\small
\begin{tabular}{|p{3.5cm}|p{11.5cm}|}
\hline
\textbf{Mã chức năng} & \textbf{FR-04: Quản lý Giỏ hàng và Tạo đơn hàng} \\ \hline
\textbf{Tác nhân (Actor)} & Khách hàng (Customer), Hệ thống CSDL (PostgreSQL) \\ \hline
\textbf{Mô tả tóm tắt} & Khách hàng quản lý giỏ hàng (thêm, sửa số lượng, xóa, nhập mã giảm giá) và bấm tạo đơn hàng. Backend kiểm tra tồn kho, áp dụng khóa lạc quan và tạo đơn hàng trạng thái \texttt{PENDING}. \\ \hline
\textbf{Tiền điều kiện} & 
\begin{enumerate}
    \vspace{-0.2cm}
    \item Giỏ hàng có ít nhất một sản phẩm hợp lệ (\texttt{items.length >= 1}).
    \item Khách hàng đã đăng nhập tài khoản hợp lệ (JWT Token hợp lệ).
    \vspace{-0.2cm}
\end{enumerate} \\ \hline
\textbf{Hậu điều kiện} & 
\begin{enumerate}
    \vspace{-0.2cm}
    \item Đơn hàng được tạo mới ở trạng thái \texttt{PENDING} với mã định danh \texttt{\#ORD-XXXXXX}.
    \item Tồn kho sản phẩm được cập nhật an toàn trong transaction CSDL.
    \item Chuyển tiếp khách hàng sang luồng thanh toán với mã \texttt{pendingOrderId}.
    \vspace{-0.2cm}
\end{enumerate} \\ \hline
\textbf{Luồng sự kiện chính} & 
\begin{enumerate}
    \vspace{-0.2cm}
    \item Khách hàng mở giỏ hàng, kiểm tra danh sách món đồ uống đã chọn.
    \item Khách hàng điều chỉnh số lượng hoặc áp dụng mã giảm giá Coupon (nếu có).
    \item Khách hàng bấm nút "Tiến hành đặt hàng".
    \item Client đóng gói payload \texttt{CreateOrderDto} và gửi request \texttt{POST /orders}.
    \item Backend xác thực JWT qua \texttt{JwtAuthGuard}, kiểm tra tính hợp lệ của DTO qua \texttt{ValidationPipe}.
    \item Backend mở CSDL Transaction: nạp giá thực tế từ DB, kiểm tra tồn kho, cập nhật trừ tồn kho theo cơ chế Khóa lạc quan (version check).
    \item Backend tạo bản ghi \texttt{Order} kèm các \texttt{OrderItem} và \texttt{OrderItemTopping}.
    \item Commit transaction thành công, Backend trả về HTTP 201 Created cùng mã đơn hàng.
    \item Client nhận mã đơn và điều hướng sang màn hình Thanh toán.
    \vspace{-0.2cm}
\end{enumerate} \\ \hline
\textbf{Luồng ngoại lệ} & 
\textbf{Exc-1: Sản phẩm hết hàng (Out of Stock)}: Backend hủy bỏ transaction, trả về HTTP 400 thông báo số lượng tồn kho không đủ. \newline
\textbf{Exc-2: Xung đột phiên bản (Concurrency Conflict)}: Hệ thống tự động retry tối đa 5 lần. Nếu vượt quá, trả về HTTP 409 yêu cầu thử lại. \\ \hline
\end{tabular}
\caption{Bảng đặc tả luồng nghiệp vụ Giỏ hàng & Tạo đơn hàng (FR-04)}
\end{table}

\subsection{Mã nguồn Frontend: Quản lý State Giỏ hàng (\texttt{cartStore.ts})}
State giỏ hàng phía Client được quản lý qua Zustand, hỗ trợ lưu trữ liên tục (\textit{persist}) và tự động tính toán tổng số lượng:

\begin{lstlisting}[caption={Zustand Cart State Store Implementation}]
// Zustand store managing reactive client cart state
export const useAppStore = create<AppStore>()(
  persist(
    (set, get) => ({
      items: [],
      addToCart: (item) => {
        const { items } = get();
        const existingIdx = items.findIndex(
          (i) => i.productId === item.productId && i.size === item.size
        );
        if (existingIdx > -1) {
          const updated = [...items];
          updated[existingIdx].quantity += item.quantity;
          updated[existingIdx].totalPrice = updated[existingIdx].unitPrice * updated[existingIdx].quantity;
          set({ items: updated });
        } else {
          set({ items: [...items, item] });
        }
      },
      updateQuantity: (cartItemId, delta) => {
        set({
          items: get().items
            .map((i) => i.cartItemId === cartItemId ? { ...i, quantity: i.quantity + delta } : i)
            .filter((i) => i.quantity > 0)
        });
      },
      removeFromCart: (cartItemId) => {
        set({ items: get().items.filter((i) => i.cartItemId !== cartItemId) });
      },
      getTotalAmount: () => get().items.reduce((sum, i) => sum + i.totalPrice, 0),
      getTotalItems: () => get().items.reduce((sum, i) => sum + i.quantity, 0),
    }),
    { name: 'brewlite-storage' }
  )
);
\end{lstlisting}

\subsection{Mã nguồn Backend: Engine Tạo đơn hàng (\texttt{orders.service.ts})}
Backend xác thực tồn kho, áp dụng khóa lạc quan và tạo đơn hàng an toàn:

\begin{lstlisting}[caption={Atomic Order Creation with Optimistic Locking}]
@Injectable()
export class OrdersService {
  constructor(private readonly prisma: PrismaService) {}

  async create(userId: string, dto: CreateOrderDto) {
    for (let attempt = 1; attempt <= MAX_RETRIES; attempt++) {
      try {
        return await this.createOnce(userId, dto);
      } catch (e) {
        if (!(e instanceof VersionConflictError)) throw e;
        if (attempt === MAX_RETRIES) {
          throw new ConflictException('System busy with concurrent orders, please retry');
        }
        await new Promise((r) => setTimeout(r, 20 + Math.random() * 60));
      }
    }
  }

  private createOnce(userId: string, dto: CreateOrderDto) {
    return this.prisma.$transaction(async (tx) => {
      // 1. Fetch real product prices from DB (Do not trust client)
      const products = await tx.product.findMany({
        where: { id: { in: dto.items.map((i) => i.productId) } },
      });
      
      // 2. Optimistic stock deduction
      for (const item of dto.items) {
        const prod = products.find((p) => p.id === item.productId);
        if (!prod || prod.stock < item.qty) {
          throw new BadRequestException('Product is out of stock');
        }
        await tx.product.update({
          where: { id: prod.id, version: prod.version },
          data: { stock: prod.stock - item.qty, version: prod.version + 1 },
        });
      }

      // 3. Create Order with PENDING status
      return tx.order.create({
        data: {
          orderNumber: `ORD-${Date.now().toString().slice(-6)}`,
          userId,
          status: 'PENDING',
          subtotal: calculatedSubtotal,
          total: finalTotal,
        },
      });
    });
  }
}
\end{lstlisting}
